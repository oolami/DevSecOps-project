def associate_status(df, column_name, ind, status_col, cust_dict):
    target_group_dict = {}

    # Read status cell safely
    raw_status = df.get(status_col, pd.Series([""] * len(df))).iloc[ind]
    if pd.isna(raw_status) or str(raw_status).strip() == "":
        status_list = []
    else:
        status_list = [s.strip() for s in str(raw_status).split(";")]

    # Read value cell safely (Target Group or LB Name)
    raw_val = df.get(column_name, pd.Series([""] * len(df))).iloc[ind]
    if pd.isna(raw_val) or str(raw_val).strip() == "":
        val_list = []
    else:
        val_list = [v.strip() for v in str(raw_val).split(";")]

    # If one side is missing, we cannot map; return empty mapping
    if not val_list:
        logging.info(f"In csv row index {ind}: no {column_name}")
        return {}

    # Normalize lengths: if status missing, treat everything as "Placeholder"
    if not status_list:
        status_list = ["Placeholder"] * len(val_list)

    # Align lengths (avoid index errors)
    n = min(len(status_list), len(val_list))
    for i in range(n):
        st = status_list[i] if status_list[i] else "Placeholder"
        target_group_dict.setdefault(st, []).append(val_list[i])

    return target_group_dict
